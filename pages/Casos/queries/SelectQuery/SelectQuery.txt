SELECT 
	c.id,
	c.expediente, 
	c.spoa,
	c.spoa_expediente,
	c.oficio, 
	c.fecha_hora_hechos, 
	c.clase_accidente,
	c.direccion_completa, 
	c.comuna,
	c.latitud,
	c.longitud,
	c.creado_por,
	c.coordinador,
	(SELECT COUNT(*) FROM public.personas p WHERE p.caso_id = c.id) AS cantidad_personas,
	(SELECT COUNT(*) FROM public.personas p WHERE p.caso_id = c.id AND p.vinculacion_caso = 'Occiso') AS cantidad_occisos,
	(SELECT COUNT(*) FROM public.personas p WHERE p.caso_id = c.id AND p.vinculacion_caso = 'Victima (Lesionado)') AS cantidad_heridos,
	(SELECT COUNT(*) FROM public.vehiculos v WHERE v.caso_id = c.id) AS cantidad_vehiculos
FROM public.caso c
WHERE 
  c.oficio IS NOT NULL
  AND (
    '{{Filtros.limpiar(filtro_oficio.text)}}' = ''
    OR c.oficio::TEXT ILIKE '%{{Filtros.limpiar(filtro_oficio.text)}}%'
  )
  AND (
    '{{filtro_fecha.formattedDate || ""}}' = ''
    OR (
      c.fecha_hora_hechos >= '{{filtro_fecha.formattedDate || "1900-01-01"}}'::date
      AND c.fecha_hora_hechos < ('{{filtro_fecha.formattedDate || "1900-01-01"}}'::date + interval '1 day')
    )
  )
ORDER BY c.fecha_hora_hechos DESC;
