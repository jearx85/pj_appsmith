SELECT 
  to_char(date_trunc('month', c.fecha_hora_levantamiento), 'YYYY-MM') AS mes,
  p.vinculacion_caso,
  COUNT(*) AS cantidad
FROM public.personas p
JOIN public.caso c ON p.caso_id = c.id
WHERE c.fecha_hora_levantamiento IS NOT NULL
  AND p.vinculacion_caso IS NOT NULL
  AND COALESCE(c.remitido::text, '') LIKE {{ !Select_Remitido.selectedOptionValue || Select_Remitido.selectedOptionValue === 'Todos' ? '%' : Select_Remitido.selectedOptionValue }}
  AND ({{ Select_MesDesde.formattedDate || null }}::timestamp IS NULL OR c.fecha_hora_levantamiento >= {{ Select_MesDesde.formattedDate }}::timestamp)
  AND ({{ Select_MesHasta.formattedDate || null }}::date IS NULL OR c.fecha_hora_levantamiento < ({{ Select_MesHasta.formattedDate }}::date + INTERVAL '1 day'))
GROUP BY 1, 2
ORDER BY 1, 2;
