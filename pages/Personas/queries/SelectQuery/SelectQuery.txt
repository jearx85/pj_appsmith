SELECT p.*, 
       c.fecha_hora_hechos, 
       c.fecha_hora_levantamiento, 
       c.lugar_inspeccion, 
       c.clase_accidente, 
       c.hipotesis, 
       c.relato,
			 c.direccion_completa,
			 c.creado_por AS caso_creado_por,
       c.fecha_creacion AS fecha_creacion_caso,
       v.clase_vehiculo
FROM public."personas" p
LEFT JOIN public."caso" c ON p.expediente = c.spoa_expediente
LEFT JOIN public."vehiculos" v 
       ON v.expediente = c.spoa_expediente
      AND TRIM(v.num_documento_relacionado) = TRIM(p.cedula)
WHERE (
    p."cedula" ILIKE '%{{FiltrosSQL.busqueda()}}%'
    OR p."expediente" ILIKE '%{{FiltrosSQL.busqueda()}}%'
)
AND c.fecha_hora_hechos::date >= COALESCE(NULLIF('{{DatePickerDesde.formattedDate || ""}}', '')::date, '1900-01-01'::date)
AND c.fecha_hora_hechos::date <= COALESCE(NULLIF('{{DatePickerHasta.formattedDate || ""}}', '')::date, '2999-12-31'::date)
AND ('{{FiltrosSQL.oficio()}}' = '' OR p.oficio::TEXT ILIKE '%{{FiltrosSQL.oficio()}}%')
AND ('{{Select_Municipio.selectedOptionValue || "TODOS"}}' = 'TODOS'
     OR ('{{Select_Municipio.selectedOptionValue || "TODOS"}}' = 'MEDELLIN' AND (LEFT(TRIM(COALESCE(p.expediente, '')), 5) = '05001' OR LEFT(TRIM(COALESCE(p.expediente, '')), 4) = '5001'))
     OR ('{{Select_Municipio.selectedOptionValue || "TODOS"}}' = 'OTROS' AND NOT (LEFT(TRIM(COALESCE(p.expediente, '')), 5) = '05001' OR LEFT(TRIM(COALESCE(p.expediente, '')), 4) = '5001')))
AND ('{{FiltrosSQL.vinculacion()}}' = 'TODOS' OR LOWER(TRANSLATE(TRIM(p.vinculacion_caso), 'ÁÉÍÓÚáéíóú', 'AEIOUaeiou')) = LOWER(TRANSLATE(TRIM('{{FiltrosSQL.vinculacion()}}'), 'ÁÉÍÓÚáéíóú', 'AEIOUaeiou')))
AND ('{{Select_Familia.selectedOptionValue || "TODOS"}}' = 'TODOS'
     OR ('{{Select_Familia.selectedOptionValue || "TODOS"}}' = 'SI' AND LOWER(TRANSLATE(TRIM(COALESCE(p.familia, '')), 'ÁÉÍÓÚáéíóú', 'AEIOUaeiou')) = 'si')
     OR ('{{Select_Familia.selectedOptionValue || "TODOS"}}' = 'NO' AND LOWER(TRANSLATE(TRIM(COALESCE(p.familia, '')), 'ÁÉÍÓÚáéíóú', 'AEIOUaeiou')) = 'no')
     OR ('{{Select_Familia.selectedOptionValue || "TODOS"}}' = 'SIN_DATO' AND LOWER(TRANSLATE(TRIM(COALESCE(p.familia, '')), 'ÁÉÍÓÚáéíóú', 'AEIOUaeiou')) NOT IN ('si', 'no')))
ORDER BY c.fecha_hora_levantamiento DESC, p.id_persona DESC
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
