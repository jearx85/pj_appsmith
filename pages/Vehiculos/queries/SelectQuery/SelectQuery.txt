SELECT *
FROM (
  SELECT DISTINCT ON (
    v."expediente",
    COALESCE(NULLIF(UPPER(TRIM(v."placa")), ''), 'SIN_PLACA_' || v."id_vehiculo")
  )
    v.*,
    c.fecha_hora_hechos,
    c.oficio,
    c.coordinador
  FROM public."vehiculos" v
  JOIN public."caso" c ON v."expediente" = c."spoa_expediente"
  WHERE (
    v."placa" ILIKE '%{{data_table.searchText || ""}}%'
    OR v."expediente" ILIKE '%{{data_table.searchText || ""}}%'
  )
  AND c."oficio" IS NOT NULL
  ORDER BY
  v."expediente",
  COALESCE(NULLIF(UPPER(TRIM(v."placa")), ''), 'SIN_PLACA_' || v."id_vehiculo"),
  c.fecha_hora_hechos DESC NULLS LAST,
  v."id_vehiculo" DESC                  -- si el vehículo está repetido, se queda con el último insertado
) t
ORDER BY
  EXTRACT(YEAR FROM COALESCE(t.fecha_hora_hechos, '1900-01-01')) DESC,
  t.oficio DESC,
  "{{data_table.sortOrder.column || 'id_vehiculo'}}" {{data_table.sortOrder.order || 'ASC'}}
--LIMIT {{data_table.pageSize}}
--OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
